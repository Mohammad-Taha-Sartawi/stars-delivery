"""Act as the test driver against the local server's real mobile API.

usage: python driver.py pending | offer <price> <minutes> | orders | start <id> | complete <id> | stats
"""
import json, os, sys, urllib.request, urllib.error

BASE = 'http://localhost:3999/api'
HERE = os.path.dirname(os.path.abspath(__file__))
TOKEN_FILE = os.path.join(HERE, '.driver-token')


def call(method, path, body=None, token=None):
    data = json.dumps(body).encode() if body is not None else None
    req = urllib.request.Request(BASE + path, data=data, method=method,
                                 headers={'Content-Type': 'application/json', **({'Authorization': 'Bearer ' + token} if token else {})})
    try:
        with urllib.request.urlopen(req) as r:
            return r.status, json.loads(r.read() or b'null')
    except urllib.error.HTTPError as e:
        return e.code, json.loads(e.read() or b'null')


def token():
    if os.path.exists(TOKEN_FILE):
        return open(TOKEN_FILE).read().strip()
    line = [l for l in open(os.path.join(HERE, 'test-accounts.txt'), encoding='utf-8') if l.startswith('provider')][0]
    pw = line.split('password=')[1].strip()
    st, body = call('POST', '/auth/login', {'email': 'driver@local.test', 'password': pw, 'role': 'provider'})
    if st != 200:
        sys.exit(f'login failed {st} {body}')
    open(TOKEN_FILE, 'w').write(body['token'])
    return body['token']


def short(o):
    return {k: o.get(k) for k in ('_id', 'status', 'type', 'description', 'price', 'area', 'completionRequestedAt') if k in o}


cmd = sys.argv[1]
t = token()
if cmd == 'pending':
    st, b = call('GET', '/provider/pending-orders', token=t)
    print(st, [short(o) for o in b] if isinstance(b, list) else b)
elif cmd == 'offer':
    st, b = call('GET', '/provider/pending-orders', token=t)
    oid = b[0]['_id']
    st, b = call('POST', '/offers', {'orderId': oid, 'price': float(sys.argv[2]), 'estimatedTime': int(sys.argv[3])}, token=t)
    print('offer', st, b)
elif cmd == 'orders':
    st, b = call('GET', '/provider/orders', token=t)
    print(st, [short(o) for o in b] if isinstance(b, list) else b)
elif cmd in ('start', 'complete'):
    path = f"/orders/{sys.argv[2]}/{'fulfilling' if cmd == 'start' else 'complete'}"
    print(cmd, *call('PUT', path, {}, token=t))
elif cmd == 'stats':
    print(*call('GET', '/provider/stats', token=t))
