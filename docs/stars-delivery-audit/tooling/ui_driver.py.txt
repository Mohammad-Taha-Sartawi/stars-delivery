"""Tiny adb/UI Automator driver for the audit emulator.

usage:
  python ui.py dump                    list visible nodes (text / content-desc / bounds)
  python ui.py tap "<label>" [n]       tap the n-th node whose text/desc contains label
  python ui.py tapxy X Y               tap device pixels
  python ui.py type "<text>"           type into the focused field (ASCII only)
  python ui.py key BACK|ENTER|...      send a keyevent
  python ui.py shot <path.png>         save a device screenshot
  python ui.py swipe up|down           scroll
"""
import os, re, subprocess, sys, time

ADB = os.path.join(os.environ['LOCALAPPDATA'], 'Android', 'Sdk', 'platform-tools', 'adb.exe')
DEV = os.environ.get('AUDIT_DEVICE', 'emulator-5556')


def adb(*args, binary=False):
    out = subprocess.run([ADB, '-s', DEV, *args], capture_output=True)
    return out.stdout if binary else out.stdout.decode('utf-8', 'replace')


def nodes():
    adb('shell', 'uiautomator', 'dump', '/sdcard/ui.xml')
    xml = adb('shell', 'cat', '/sdcard/ui.xml')
    res = []
    for m in re.finditer(r'<node [^>]*>', xml):
        n = m.group(0)
        get = lambda k: (re.search(k + r'="([^"]*)"', n) or [None, ''])[1]
        b = re.findall(r'\d+', get('bounds'))
        if len(b) != 4:
            continue
        label = (get('text') + ' | ' + get('content-desc')).strip(' |')
        res.append({'label': label.replace('&#10;', ' / ').replace('&amp;', '&'), 'cls': get('class').split('.')[-1],
                    'bounds': list(map(int, b)), 'click': get('clickable') == 'true'})
    return res


def main():
    cmd = sys.argv[1]
    if cmd == 'dump':
        for n in nodes():
            if n['label'] or n['cls'] == 'EditText':
                print(f"{n['cls']:<14} {'C' if n['click'] else ' '} {n['bounds']} {n['label'][:110]}")
    elif cmd == 'tap':
        want = sys.argv[2]
        idx = int(sys.argv[3]) if len(sys.argv) > 3 else 0
        hits = [n for n in nodes() if want in n['label']]
        if len(hits) <= idx:
            print('NOT FOUND:', want)
            sys.exit(1)
        x1, y1, x2, y2 = hits[idx]['bounds']
        adb('shell', 'input', 'tap', str((x1 + x2) // 2), str((y1 + y2) // 2))
        print('tapped', hits[idx]['label'][:80])
    elif cmd == 'tapxy':
        adb('shell', 'input', 'tap', sys.argv[2], sys.argv[3])
    elif cmd == 'type':
        adb('shell', 'input', 'text', sys.argv[2].replace(' ', '%s'))
    elif cmd == 'key':
        adb('shell', 'input', 'keyevent', 'KEYCODE_' + sys.argv[2])
    elif cmd == 'shot':
        time.sleep(float(os.environ.get('SHOT_DELAY', '0.8')))
        open(sys.argv[2], 'wb').write(adb('exec-out', 'screencap', '-p', binary=True))
        print('saved', sys.argv[2])
    elif cmd == 'swipe':
        if sys.argv[2] == 'up':
            adb('shell', 'input', 'swipe', '540', '1800', '540', '700', '300')
        else:
            adb('shell', 'input', 'swipe', '540', '700', '540', '1800', '300')


main()
