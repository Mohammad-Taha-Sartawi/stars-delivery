// Local test environment for Stars Delivery (audit/QA only).
// Real backend code + throwaway in-memory MongoDB *replica set* (transactions
// work), every env var overridden so nothing touches real services.
// usage: node server.js <backendDir> <outDir>
const path = require('path');
const fs = require('fs');
const crypto = require('crypto');
const os = require('os');

const BACKEND = process.argv[2];
const OUT = process.argv[3];
const req = (m) => require(path.join(BACKEND, 'node_modules', m));
const { MongoMemoryReplSet } = req('mongodb-memory-server-core');

(async () => {
  const rs = await MongoMemoryReplSet.create({
    replSet: { count: 1, storageEngine: 'wiredTiger' },
    binary: {
      version: '8.2.6',
      downloadDir: path.join(os.tmpdir(), 'swifora-mongodb-binaries'),
    },
  });
  const rnd = () => crypto.randomBytes(32).toString('hex');
  const docs = path.join(OUT, 'provider-docs');
  fs.mkdirSync(docs, { recursive: true });
  Object.assign(process.env, {
    NODE_ENV: 'development',
    PORT: '3999',
    MONGODB_URI: rs.getUri('stars_local'),
    JWT_SECRET: rnd(),
    SESSION_SECRET: rnd(),
    PASSWORD_RECOVERY_SECRET: rnd(),
    CORS_ALLOWED_ORIGINS: 'http://localhost:3999',
    IMGBB_API_KEY: '',
    SMTP_HOST: '127.0.0.1',
    SMTP_PORT: '1',
    SMTP_SECURE: 'false',
    SMTP_USER: 'local',
    SMTP_PASS: 'local',
    SMTP_FROM: 'local@example.test',
    PROVIDER_DOCUMENTS_DIR: docs,
    FIREBASE_SERVICE_ACCOUNT_JSON: '',
    FIREBASE_SERVICE_ACCOUNT_PATH: '',
    // The Android emulator reports mocked GPS fixes.
    ALLOW_MOCK_LOCATION: 'true',
  });

  require(path.join(BACKEND, 'src', 'index.js'));
  const mongoose = req('mongoose');
  const bcrypt = req('bcryptjs');
  while (mongoose.connection.readyState !== 1) {
    await new Promise((r) => setTimeout(r, 200));
  }
  const User = require(path.join(BACKEND, 'src/models/User'));
  const adminPassword = `Local-${crypto.randomBytes(9).toString('base64url')}`;
  await User.create({
    fullName: 'Local Admin',
    email: 'admin@local.test',
    password: await bcrypt.hash(adminPassword, 12),
    role: 'admin',
    status: 'active',
  });
  fs.appendFileSync(
    path.join(OUT, 'test-accounts.txt'),
    `admin email=admin@local.test password=${adminPassword}\n`,
  );
  console.log('LOCAL_ENV_READY http://localhost:3999');
})().catch((e) => {
  console.error('LOCAL_ENV_FAILED', e);
  process.exit(1);
});
