// Audit-only: boots the real backend against an in-memory MongoDB seeded with
// synthetic data so the admin dashboard can be screenshotted. Nothing here
// touches the real database: every env var the backend reads is overridden
// before src/index.js (and its dotenv call) loads.
const path = require('path');
const fs = require('fs');
const crypto = require('crypto');
const os = require('os');

const BACKEND = process.argv[2];
const OUT = process.argv[3];
const req = (m) => require(path.join(BACKEND, 'node_modules', m));
const { MongoMemoryServer } = req('mongodb-memory-server-core');

(async () => {
  const mongo = await MongoMemoryServer.create({
    binary: {
      version: '8.2.6',
      downloadDir: path.join(os.tmpdir(), 'stars-delivery-mongodb-binaries'),
    },
  });
  const rnd = () => crypto.randomBytes(32).toString('hex');
  const docs = path.join(OUT, 'provider-docs');
  fs.mkdirSync(docs, { recursive: true });
  Object.assign(process.env, {
    NODE_ENV: 'development',
    PORT: '3999',
    MONGODB_URI: mongo.getUri('stars_audit'),
    JWT_SECRET: rnd(),
    SESSION_SECRET: rnd(),
    PASSWORD_RECOVERY_SECRET: rnd(),
    CORS_ALLOWED_ORIGINS: 'http://localhost:3999',
    IMGBB_API_KEY: '',
    SMTP_HOST: '127.0.0.1',
    SMTP_PORT: '1',
    SMTP_SECURE: 'false',
    SMTP_USER: 'audit',
    SMTP_PASS: 'audit',
    SMTP_FROM: 'audit@example.test',
    PROVIDER_DOCUMENTS_DIR: docs,
    FIREBASE_SERVICE_ACCOUNT_JSON: '',
    FIREBASE_SERVICE_ACCOUNT_PATH: '',
    ALLOW_MOCK_LOCATION: 'false',
  });

  require(path.join(BACKEND, 'src', 'index.js'));
  const mongoose = req('mongoose');
  const bcrypt = req('bcryptjs');
  while (mongoose.connection.readyState !== 1) {
    await new Promise((r) => setTimeout(r, 200));
  }
  const User = require(path.join(BACKEND, 'src/models/User'));
  const Order = require(path.join(BACKEND, 'src/models/Order'));
  const Offer = require(path.join(BACKEND, 'src/models/Offer'));

  const adminPassword = `Audit-${crypto.randomBytes(9).toString('base64url')}`;
  fs.writeFileSync(
    path.join(OUT, 'admin-credentials.txt'),
    `email=admin@example.test\npassword=${adminPassword}\n`,
  );
  await User.create({
    fullName: 'Audit Admin',
    email: 'admin@example.test',
    password: await bcrypt.hash(adminPassword, 12),
    role: 'admin',
    status: 'active',
  });
  const pw = await bcrypt.hash(`x-${rnd()}`, 4);
  const mk = (i, role, extra = {}) =>
    User.create({
      fullName: `${role === 'provider' ? 'Driver' : 'Customer'} ${i}`,
      email: `${role}${i}@example.test`,
      phoneNumbers: [{ number: `05900000${i}${i}` }],
      password: pw,
      role,
      area: 'رام الله',
      governorate: 'رام الله والبيرة',
      privacyPolicy: true,
      ...extra,
    });
  const customers = [];
  for (let i = 1; i <= 4; i++) customers.push(await mk(i, 'customer'));
  const drivers = [];
  for (let i = 1; i <= 3; i++) {
    drivers.push(
      await mk(i, 'provider', {
        serviceTypes: ['product', 'people'],
        status: i === 3 ? 'pending' : 'active',
        verified: i !== 3,
      }),
    );
  }
  const statuses = ['pending', 'offered', 'accepted', 'fulfilling', 'completed', 'cancelled', 'expired'];
  let n = 0;
  for (const status of statuses) {
    n++;
    const assigned = ['accepted', 'fulfilling', 'completed'].includes(status);
    const order = await Order.create({
      customerId: customers[n % customers.length]._id,
      providerId: assigned ? drivers[n % 2]._id : undefined,
      type: ['product', 'people', 'goods'][n % 3],
      description: `Synthetic audit order #${n} (${status})`,
      phone: '0590000000',
      area: 'رام الله',
      governorate: 'رام الله والبيرة',
      status,
      price: assigned ? 10 + n : 0,
      platformCommission: status === 'completed' ? 1.5 : 0,
      providerEarning: status === 'completed' ? 10 + n - 1.5 : 0,
      completedAt: status === 'completed' ? new Date() : null,
    });
    if (status === 'offered') {
      await Offer.create({ orderId: order._id, providerId: drivers[0]._id, price: 14, estimatedTime: 20, expiresAt: new Date(Date.now() + 5 * 60000) });
    }
  }
  console.log('AUDIT_DEMO_READY http://localhost:3999/admin');
})().catch((e) => {
  console.error('AUDIT_DEMO_FAILED', e.message);
  process.exit(1);
});
