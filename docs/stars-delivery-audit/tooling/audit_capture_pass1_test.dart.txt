class AuditAdapter implements HttpClientAdapter {
  Mode mode = Mode.populated;
  Duration delay = Duration.zero;

  dynamic _body(String method, String path) {
    final p = path.replaceFirst(RegExp(r'^/api'), '');
    final empty = mode == Mode.empty;
    if (p.endsWith('/unread-count')) return {'count': empty ? 0 : 2};
    if (p == '/provider/stats') return _stats;
    if (p == '/ads') return empty ? [] : _ads;
    if (p == '/users/available-providers') return {'count': empty ? 0 : 7};
    if (p == '/notifications') return empty ? [] : _notifications;
    if (p == '/chat/history') return empty ? [] : _chat;
    if (p == '/reports/my') return [];
    if (p == '/provider/pending-orders') return empty ? [] : _pendingForProvider;
    if (p == '/provider/offered-orders') return empty ? [] : _offeredForProvider;
    if (p == '/provider/orders') return empty ? [] : _activeForProvider;
    if (p == '/provider/rejected-orders') return empty ? [] : _rejected;
    if (p.startsWith('/offers/order/')) return empty ? [] : _offers;
    if (p == '/orders' && method == 'GET') return empty ? [] : _myOrders;
    if (p.startsWith('/orders/') && method == 'GET') {
      return {
        ..._order('ord-1001', 'fulfilling', price: 15),
        'providerName': 'Driver Alpha',
        'providerLocation': {'lat': 31.905, 'lng': 35.205},
        'myOffer': null,
      };
    }
    return {'ok': true};
  }

  @override
  Future<ResponseBody> fetch(
    RequestOptions options,
    Stream<Uint8List>? requestStream,
    Future<void>? cancelFuture,
  ) async {
    if (delay > Duration.zero) await Future<void>.delayed(delay);
    final failing = mode == Mode.error;
    final data = failing
        ? {'error': 'Synthetic server error (audit fixture)'}
        : _body(options.method, options.uri.path);
    return ResponseBody.fromString(
      jsonEncode(data),
      failing ? 500 : 200,
      headers: {
        Headers.contentTypeHeader: [Headers.jsonContentType],
      },
    );
  }

  @override
  void close({bool force = false}) {}
}

Future<void> _loadFonts() async {
  Future<ByteData> f(String p) async =>
      ByteData.view(Uint8List.fromList(await File(p).readAsBytes()).buffer);
  final roboto = FontLoader('Roboto')
    ..addFont(f(r'C:\Windows\Fonts\segoeui.ttf'))
    ..addFont(f(r'C:\Windows\Fonts\segoeuib.ttf'));
  await roboto.load();
  final icons = FontLoader('MaterialIcons')
    ..addFont(
      f(r'C:\src\flutter\bin\cache\artifacts\material_fonts\materialicons-regular.otf'),
    );
  await icons.load();
}

Future<void> _snap(WidgetTester tester, GlobalKey key, String name, String note) async {
  final boundary = key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
  final image = await tester.runAsync(() => boundary.toImage(pixelRatio: 2.0));
  final bytes = await tester.runAsync(
    () => image!.toByteData(format: ui.ImageByteFormat.png),
  );
  final file = File('$_out${Platform.pathSeparator}$name.png');
  await tester.runAsync(() => file.writeAsBytes(bytes!.buffer.asUint8List()));
  _manifest.add({
    'file': '$name.png',
    'width': '${image!.width}',
    'height': '${image.height}',
    'note': note,
  });
}

const _variants = [
  ('ar-light', Locale('ar'), ThemeMode.light),
  ('en-light', Locale('en'), ThemeMode.light),
  ('ar-dark', Locale('ar'), ThemeMode.dark),
];

void _viewport(WidgetTester tester) {
  tester.view.physicalSize = const Size(1170, 2532); // 390x844 logical @3x
  tester.view.devicePixelRatio = 3.0;
}

Future<void> main() async {
  if (Platform.environment['STARS_AUDIT_CAPTURE'] != '1') {
    test('audit capture disabled', () {});
    return;
  }
  TestWidgetsFlutterBinding.ensureInitialized();
  await _loadFonts();
  Directory(_out).createSync(recursive: true);
  final harness = Ux03Harness();
  final adapter = AuditAdapter();
  MobileApiClient.testAdapter = adapter;

  setUp(() {
    SharedPreferences.setMockInitialValues({});
    adapter
      ..mode = Mode.populated
      ..delay = Duration.zero;
    MobileApiClient.testAdapter = adapter;
  });

  Future<void> pumpCase(
    WidgetTester tester,
    Widget home,
    GlobalKey key, {
    Locale locale = const Locale('ar'),
    ThemeMode theme = ThemeMode.light,
    bool settle = true,
  }) async {
    _viewport(tester);
    await tester.pumpWidget(harness.app(home, themeMode: theme, locale: locale, boundaryKey: key));
    await tester.pump();
    if (settle) {
      for (var i = 0; i < 6; i++) {
        await tester.pump(const Duration(milliseconds: 200));
      }
    }
  }

  Future<void> drain(WidgetTester tester) async {
    await tester.pumpWidget(const SizedBox.shrink());
    for (var i = 0; i < 20; i++) {
      await tester.pump(const Duration(seconds: 1));
    }
    tester.takeException();
  }

  // Login (real app shell with stubbed auth gateway).
  for (final (vName, locale, theme) in _variants) {
    testWidgets('login $vName', (tester) async {
      _viewport(tester);
      final key = GlobalKey();
      await tester.pumpWidget(
        RepaintBoundary(
          key: key,
          child: StarsDeliveryApp(
            initialLocale: locale,
            initialTheme: theme,
            authBlocFactory: () => AuthBloc(
              authService: _Gateway(),
              sessionStorage: SessionStorage(vault: _Vault()),
              checkOnStart: false,
            ),
          ),
        ),
      );
      for (var i = 0; i < 5; i++) {
        await tester.pump(const Duration(milliseconds: 200));
      }
      await _snap(tester, key, 'login-$vName-default', 'StarsDeliveryApp -> LoginScreen');
      if (vName == 'ar-light') {
        // Submit empty form to surface validation errors.
        final buttons = find.byType(ElevatedButton);
        if (buttons.evaluate().isNotEmpty) {
          await tester.ensureVisible(buttons.first);
          await tester.tap(buttons.first, warnIfMissed: false);
          await tester.pump(const Duration(milliseconds: 300));
          await _snap(tester, key, 'login-$vName-validation', 'Tapped first ElevatedButton with empty fields');
        }
      }
      await drain(tester);
    });
  }

  for (final c in _cases) {
    for (final (vName, locale, theme) in _variants) {
      testWidgets('${c.id} $vName', (tester) async {
        final key = GlobalKey();
        await pumpCase(tester, c.build(), key, locale: locale, theme: theme);
        await _snap(tester, key, '${c.id}-$vName-populated', 'populated fixtures');
        await drain(tester);
      });
    }
    if (c.list) {
      testWidgets('${c.id} empty', (tester) async {
        adapter.mode = Mode.empty;
        final key = GlobalKey();
        await pumpCase(tester, c.build(), key);
        await _snap(tester, key, '${c.id}-ar-light-empty', 'empty fixtures');
        await drain(tester);
      });
      testWidgets('${c.id} error', (tester) async {
        adapter.mode = Mode.error;
        final key = GlobalKey();
        await pumpCase(tester, c.build(), key);
        await _snap(tester, key, '${c.id}-ar-light-error', 'HTTP 500 for every request');
        await drain(tester);
      });
      testWidgets('${c.id} loading', (tester) async {
        adapter.delay = const Duration(seconds: 3);
        final key = GlobalKey();
        await pumpCase(tester, c.build(), key, settle: false);
        await tester.pump(const Duration(milliseconds: 300));
        await _snap(tester, key, '${c.id}-ar-light-loading', 'responses delayed 3s; captured at 300ms');
        await drain(tester);
      });
    }
  }

  // Dialog: logout confirmation from customer home (en + ar).
  for (final (vName, locale, theme) in _variants) {
    testWidgets('logout dialog $vName', (tester) async {
      final key = GlobalKey();
      await pumpCase(
        tester,
        HomeScreen(user: _cust, role: 'customer', token: _t),
        key,
        locale: locale,
        theme: theme,
      );
      final logout = find.byIcon(Icons.logout);
      if (logout.evaluate().isNotEmpty) {
        await tester.tap(logout.first, warnIfMissed: false);
        for (var i = 0; i < 4; i++) {
          await tester.pump(const Duration(milliseconds: 150));
        }
        await _snap(tester, key, 'dialog-logout-$vName', 'tapped Icons.logout on HomeScreen');
      }
      await drain(tester);
    });
  }

  tearDownAll(() {
    File('$_out${Platform.pathSeparator}_capture_manifest.json')
        .writeAsStringSync(const JsonEncoder.withIndent('  ').convert(_manifest));
    harness.dispose();
  });
}
